cmake_minimum_required(VERSION 3.16)
project(JC3xGTA5 CXX)
set(SHV_SDK "${CMAKE_SOURCE_DIR}/sdk" CACHE PATH "ScriptHookV SDK root (contains inc/ and lib/)")
find_package(Python3 REQUIRED)
add_custom_target(gen ALL COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/../codegen/gen.py)
add_library(JC3xGTA5 SHARED src/main.cpp)
add_dependencies(JC3xGTA5 gen)
set_target_properties(JC3xGTA5 PROPERTIES SUFFIX ".asi" CXX_STANDARD 17)
target_include_directories(JC3xGTA5 PRIVATE ${SHV_SDK}/inc)
target_link_libraries(JC3xGTA5 PRIVATE ${SHV_SDK}/lib/ScriptHookV.lib)
