cmake_minimum_required(VERSION 3.20)
project(JC3xGTA5_jc3 CXX C)
include(FetchContent)
FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git GIT_TAG v1.3.4)
FetchContent_Declare(reshade GIT_REPOSITORY https://github.com/crosire/reshade.git GIT_TAG v6.8.0 GIT_SHALLOW TRUE
  SOURCE_SUBDIR do-not-build)
FetchContent_MakeAvailable(minhook reshade)
find_package(Python3 REQUIRED)
add_custom_target(gen_jc3 ALL COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/../codegen/gen.py)
add_library(JC3xGTA5_jc3 SHARED src/addon.cpp src/symbols.cpp)
set_target_properties(JC3xGTA5_jc3 PROPERTIES OUTPUT_NAME "JC3xGTA5" SUFFIX ".addon64" CXX_STANDARD 20
  MSVC_RUNTIME_LIBRARY "MultiThreaded")
target_compile_definitions(JC3xGTA5_jc3 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(JC3xGTA5_jc3 PRIVATE ${reshade_SOURCE_DIR}/include)
target_link_libraries(JC3xGTA5_jc3 PRIVATE minhook)
add_dependencies(JC3xGTA5_jc3 gen_jc3)
